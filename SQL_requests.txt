Запрос 2
SELECT
    `Order Date`,
    SUM(IF(`Customer Type`=1,`Price`*`Quantity`,0)) AS `Wholesale Revenue`,
    SUM(IF(`Customer Type`=0,`Price`*`Quantity`,0)) AS `Retail Revenue`
FROM
    orders
INNER JOIN goods ON orders.Barcode = goods.Barcode
INNER JOIN customers ON orders.`Customer ID` = customers.`Customer ID`
GROUP BY
    `Order Date`
